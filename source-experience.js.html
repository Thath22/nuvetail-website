let experienceCleanup=()=>{}, pendingChapter=null;
function bindExperience(){
 const abort=new AbortController(),signal=abort.signal,observers=[];
 const reduced=matchMedia('(prefers-reduced-motion: reduce)'), hover=matchMedia('(hover:hover) and (pointer:fine)');
 const jump=id=>{const target=document.getElementById(id);if(target){target.scrollIntoView({behavior:reduced.matches?'instant':'smooth',block:'start'});target.setAttribute('tabindex','-1');target.focus({preventScroll:true});}};
 document.querySelectorAll('[data-jump]').forEach(b=>b.addEventListener('click',()=>jump(b.dataset.jump),{signal}));
 document.querySelectorAll('[data-on-route-jump]').forEach(b=>b.addEventListener('click',()=>{pendingChapter=b.dataset.onRouteJump;},{signal}));
 function tabs(selector,panelPrefix,vertical=false){const buttons=[...document.querySelectorAll(selector)];const select=(b,focus=false)=>{buttons.forEach((x,i)=>{const active=x===b;x.setAttribute('aria-selected',String(active));x.tabIndex=active?0:-1;document.getElementById(panelPrefix+i).hidden=!active});if(focus)b.focus();};buttons.forEach((b,i)=>{b.addEventListener('click',()=>select(b),{signal});b.addEventListener('keydown',e=>{let n=i;const next=vertical?'ArrowDown':'ArrowRight',prev=vertical?'ArrowUp':'ArrowLeft';if(e.key===next)n=(i+1)%buttons.length;else if(e.key===prev)n=(i+buttons.length-1)%buttons.length;else if(e.key==='Home')n=0;else if(e.key==='End')n=buttons.length-1;else return;e.preventDefault();select(buttons[n],true)},{signal})});}
 tabs('[data-ingredient]','ingredient-panel-');tabs('[data-ritual]','ritual-panel-',true);
 const sticky=document.querySelector('.sticky-bundle'),purchase=document.querySelector('.purchase');
 if(sticky){syncStickyBundle();document.getElementById('sticky-add').addEventListener('click',()=>document.getElementById('add').click(),{signal});}
 const reveal=new IntersectionObserver(entries=>{entries.forEach(e=>{if(e.isIntersecting){e.target.classList.add('is-revealed');reveal.unobserve(e.target)}})},{threshold:.07});observers.push(reveal);
 if(!reduced.matches)document.querySelectorAll('.love-section,.shop-bundle,.natural-ingredients-heading,.ingredient-tabs,.ingredient-panels,.ritual-layout,.ritual-section .section-head,.product-story-banner,.shared-moment,.compact-faq .section-head,.final-cta').forEach((el,i)=>{el.classList.add('scroll-reveal');el.style.setProperty('--reveal-delay',`${Math.min(i%3,2)*65}ms`);reveal.observe(el)});
 const chapters=[...document.querySelectorAll('.product-chapters [data-jump]')];
 let frame=0;
 const update=()=>{frame=0;if(sticky&&purchase)sticky.hidden=purchase.getBoundingClientRect().bottom>0;let active=chapters[0];chapters.forEach(b=>{const el=document.getElementById(b.dataset.jump);if(el&&el.getBoundingClientRect().top<innerHeight*.35)active=b});chapters.forEach(b=>b.setAttribute('aria-current',b===active?'location':'false'));};
 const schedule=()=>{if(!frame)frame=requestAnimationFrame(update)};window.addEventListener('scroll',schedule,{passive:true,signal});window.addEventListener('resize',schedule,{passive:true,signal});update();
 const hero=document.querySelector('.hero-pack-panel');
 if(hero&&hover.matches){hero.addEventListener('pointermove',e=>{if(reduced.matches)return;const r=hero.getBoundingClientRect();hero.style.setProperty('--hero-x',`${(e.clientX-r.left-r.width/2)*.018}px`);hero.style.setProperty('--hero-y',`${(e.clientY-r.top-r.height/2)*.012}px`)},{passive:true,signal});hero.addEventListener('pointerleave',()=>{hero.style.setProperty('--hero-x','0px');hero.style.setProperty('--hero-y','0px')},{signal});}
 reduced.addEventListener('change',()=>{if(reduced.matches)document.querySelectorAll('.scroll-reveal').forEach(el=>el.classList.add('is-revealed'))},{signal});
 experienceCleanup=()=>{abort.abort();observers.forEach(o=>o.disconnect());cancelAnimationFrame(frame)};
 if(pendingChapter){const id=pendingChapter;pendingChapter=null;requestAnimationFrame(()=>jump(id));}
}
